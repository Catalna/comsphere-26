```js
import assert from "node:assert/strict";
import test from "node:test";
import request from "supertest";
import { createApp } from "../src/app.js";

test("health route reports the API service", async () => {
  const response = await request(createApp()).get("/api/health");
  assert.equal(response.status, 200);
  assert.deepEqual(response.body, { status: "ok", service: "dana-terang-api" });
});

test("malformed and oversized JSON receive generic bounded errors", async () => {
  const app = createApp();
  const malformed = await request(app).post("/api/unknown")
    .set("content-type", "application/json").send("{");
  assert.deepEqual(malformed.body, { error: "invalid_json" });
  const large = await request(app).post("/api/unknown")
    .set("content-type", "application/json")
    .send(JSON.stringify({ value: "x".repeat(1_048_600) }));
  assert.deepEqual(large.body, { error: "request_too_large" });
  assert.equal(large.status, 413);
});
```
